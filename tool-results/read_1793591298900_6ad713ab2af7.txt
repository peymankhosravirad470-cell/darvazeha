   595→function drawPseudoQR(seed){
   596→  const cv = document.getElementById('qr-canvas'), ctx = cv.getContext('2d');
   597→  const N = 25, cell = Math.floor(cv.width / (N + 1)), off = cell;
   598→  let h = 2166136261;
   599→  for(const ch of seed){ h ^= ch.charCodeAt(0); h = Math.imul(h, 16777619); }
   600→  const rnd = () => { h ^= h << 13; h ^= h >>> 17; h ^= h << 5; return ((h >>> 0) / 4294967296); };
   601→  ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, cv.width, cv.height);
   602→  ctx.fillStyle = '#10132b';
   603→  for(let y = 0; y < N; y++) for(let x = 0; x < N; x++){
   604→    const inFinder = (x < 8 && y < 8) || (x >= N - 8 && y < 8) || (x < 8 && y >= N - 8);
   605→    if(inFinder) continue;
   606→    if(rnd() > .52) ctx.fillRect(off + x * cell, off + y * cell, cell - .8, cell - .8);
   607→  }
   608→  const finder = (fx, fy) => {
   609→    ctx.fillStyle = '#10132b';
   610→    ctx.fillRect(off + fx * cell, off + fy * cell, 7 * cell, 7 * cell);
   611→    ctx.fillStyle = '#fff';
   612→    ctx.fillRect(off + (fx + 1) * cell, off + (fy + 1) * cell, 5 * cell, 5 * cell);
   613→    ctx.fillStyle = '#10132b';
   614→    ctx.fillRect(off + (fx + 2) * cell, off + (fy + 2) * cell, 3 * cell, 3 * cell);
   615→  };
   616→  finder(0, 0); finder(N - 7, 0); finder(0, N - 7);
   617→}

   619→/* ---------- exploration scene ---------- */
   620→const scene = document.getElementById('scene');

   623→function enterScene(){
   624→  scene.classList.add('active');

   626→  px = 20; py = 52; place();

   630→  scene.classList.remove('active');
   631→  document.querySelector('.topbar').style.display = '';
   632→});
   633→function place(){ player.style.left = px + '%'; player.style.top = py + '%'; }
   634→
   635→/* joystick */
   636→const joy = document.getElementById('joy'), thumb = document.getElementById('joy-thumb');
   637→let joyVec = {x:0, y:0}, joyId = null;
   638→function joyMove(e){
   639→  const r = joy.getBoundingClientRect();
   640→  let dx = e.clientX - (r.left + r.width / 2), dy = e.clientY - (r.top + r.height / 2);
   641→  const dist = Math.hypot(dx, dy), max = r.width / 2 - 18;
   642→  if(dist > max){ dx *= max / dist; dy *= max / dist; }
   643→  thumb.style.transform = `translate(calc(-50% + ${dx}px), calc(-50% + ${dy}px))`;
   644→  joyVec = {x: dx / max, y: dy / max};
   645→}
   646→joy.addEventListener('pointerdown', e => {
   647→  joyId = e.pointerId; joy.classList.add('active'); joy.setPointerCapture(joyId); joyMove(e);
   648→});
   649→joy.addEventListener('pointermove', e => { if(e.pointerId === joyId) joyMove(e); });
   650→['pointerup', 'pointercancel'].forEach(ev => joy.addEventListener(ev, e => {
   651→  if(e.pointerId !== joyId) return;
   652→  joyId = null; joy.classList.remove('active');
   653→  joyVec = {x:0, y:0}; thumb.style.transform = 'translate(-50%,-50%)';
   654→}));
   655→/* keyboard */
   656→const keys = {};
   657→addEventListener('keydown', e => {
   658→  if(!scene.classList.contains('active')) return;
   659→  keys[e.key.toLowerCase()] = true;
   660→  if(['arrowup','arrowdown','arrowleft','arrowright',' '].includes(e.key.toLowerCase())) e.preventDefault();
   661→});
   662→addEventListener('keyup', e => keys[e.key.toLowerCase()] = false);
   663→
   664→(function loop(){
   665→  let dx = joyVec.x, dy = joyVec.y;
   666→  if(keys['arrowleft'] || keys['a']) dx -= 1;
   667→  if(keys['arrowright'] || keys['d']) dx += 1;
   668→  if(keys['arrowup'] || keys['w']) dy -= 1;
   669→  if(keys['arrowdown'] || keys['s']) dy += 1;
   670→  if(dx || dy){
   671→    px = Math.min(94, Math.max(6, px + dx * .35));
   672→    py = Math.min(76, Math.max(26, py + dy * .35));
   673→    place();
   674→  }
   675→  requestAnimationFrame(loop);
   676→})();

   678→/* bubbles */
   679→function spawnBubbles(){
   680→  if(scene.querySelector('.bubble')) return;
   681→  for(let i = 0; i < 16; i++){
   682→    const b = document.createElement('span');
   683→    b.className = 'bubble';
   684→    const s = 5 + Math.random() * 16;
   685→    b.style.width = b.style.height = s + 'px';
   686→    b.style.left = (Math.random() * 98) + '%';
   687→    b.style.animationDuration = (9 + Math.random() * 12) + 's';
   688→    b.style.animationDelay = (-Math.random() * 14) + 's';
   689→    scene.appendChild(b);
   690→  }
   691→}
